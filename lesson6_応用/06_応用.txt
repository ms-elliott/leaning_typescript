[077. narrowing 3つのtype guardで型を絞り込む]
    // TypeScriptは文脈を理解して、自動的に型を絞り込む
    let foo: string | number
    foo = 'hello'
    function myFunc(x: string | null) {
        // 条件式等で特定の型のみに絞り込んでいることを、type guardと呼ぶ
        if(x === null) {
            x   // <- nullしか入らないので、自動的にx: nullとなる
        } else {
            x   // <- nullは入らないので自動的にx: stringとなる
        }
    }

    // returnの場合も同様で、自動的に型を絞り込む
    function myFunc(x: string | null) {
        if(x === null) {
            return x   // x: null
        }
        return x   // x: string
    }

    3つのtype guard
    ①typeof演算子
    ②in演算子
    ③instanceof演算子

[079. as(型アサーション)　手動で型を上書きする]
    ※HTMLファイルにinputタグのついたコンソールがあるとする
    // 型アサーションをしないと、型はHTMLInputElement or null となり、.valueでエラーになる
    const input = document.getElementById('input');
    input.value = 'initial input value';

    // 型アサーションの仕方２パターン
    const input = <HTMLInputElement>document.getElementById('input')
    const input = document.getElementById('input') as HTMLInputElement;

    // .value上書きの別記法   ()で囲む
    (document.getElementById('input') as HTMLInputElement).value = 'initial input value';
